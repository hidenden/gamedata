def search(query, *, kind=None, limit=10): 
 72def search(query, *, kind=None, limit=10):
 73    """Search Japanese descriptions or Python names; bounded and deterministic.
 74
 75    Exact matches rank first, followed by token/Japanese bigram overlap.
 76    An empty query lists entries, optionally restricted by kind.
 77    """
 78    if kind is not None and kind not in KINDS:
 79        raise ValueError(f"kind must be one of {KINDS}")
 80    if not isinstance(query, str):
 81        raise TypeError("query must be str")
 82    if type(limit) is not int or not 1 <= limit <= 100:
 83        raise ValueError("limit must be between 1 and 100")
 84    query = query.strip().casefold()
 85    tokens = re.findall(r"[a-z0-9_]+|[^\W\d_a-z]+", query)
 86    fragments = set()
 87    for token in tokens:
 88        if token.isascii() or len(token) < 2:
 89            fragments.add(token)
 90        else:
 91            fragments.update(token[i : i + 2] for i in range(len(token) - 1))
 92    matches = []
 93    for identifier, entry_kind, summary in _entries():
 94        if kind is not None and entry_kind != kind:
 95            continue
 96        haystack = f"{identifier} {summary}".casefold()
 97        score = (100 if query and query in haystack else 0) + sum(
 98            f in haystack for f in fragments
 99        )
100        if not query or score:
101            matches.append(
102                {
103                    "id": identifier,
104                    "kind": entry_kind,
105                    "summary": summary,
106                    "score": score,
107                }
108            )
109    matches.sort(key=lambda item: (-item["score"], item["id"]))
110    return {
111        "schema_version": SCHEMA_VERSION,
112        "query": query,
113        "total": len(matches),
114        "items": matches[:limit],
115        "truncated": len(matches) > limit,
116    }

def describe(id, *, sections=None, params=None): 
16def describe(id, *, sections=None, params=None):
17    """Return a fresh record. params resolves function output variants only.
18
19    Unknown IDs raise KeyError; invalid parameters/sections raise ValueError.
20    No function is executed and no database is read.
21    """
22    if not isinstance(id, str) or ":" not in id:
23        raise KeyError(id)
24    kind, name = id.split(":", 1)
25    if params is not None and kind != "function":
26        raise ValueError("params is only supported for function records")
27    if kind == "dataset":
28        result = DATASETS[name]
29    elif kind == "column":
30        ds, sep, col = name.partition(".")
31        if not sep:
32            raise KeyError(id)
33        fields = DATASETS[ds]["columns"][col]
34        result = record(
35            id,
36            fields["summary"],
37            dataset=f"dataset:{ds}",
38            **{k: v for k, v in fields.items() if k != "summary"},
39        )
40    elif kind == "function":
41        result = describe_function(name, params)
42    elif kind == "recipe":
43        result = RECIPES[name]
44    elif kind == "guide" and name == "annotation_levels":
45        result = annotation_level_policy()
46    else:
47        raise KeyError(id)
48    if sections is not None:
49        if isinstance(sections, str):
50            sections = [sections]
51        invalid = set(sections) - set(result)
52        if invalid:
53            raise ValueError(f"Unknown sections: {sorted(invalid)}")
54        keep = {"schema_version", "id", "kind", "summary", *sections}
55        result = {k: v for k, v in result.items() if k in keep}
56    return deepcopy(result)

def inspect_frame(df, *, dataset=None, profile='basic', sample_rows=0): 
 12def inspect_frame(df, *, dataset=None, profile="basic", sample_rows=0):
 13    """Inspect schema and coverage, optionally testing keys/nulls/weekly gaps.
 14
 15    dataset is an explicitly supplied contract, not a provenance assertion.
 16    basic scans date and dimension columns; full can scan the entire frame.
 17    Samples are opt-in (0..20). Dimension and group listings are capped at 20.
 18    LazyFrame is rejected to avoid implicit query execution.
 19    """
 20    if not isinstance(df, pl.DataFrame):
 21        raise TypeError(
 22            "df must be a polars.DataFrame; LazyFrame is not collected implicitly"
 23        )
 24    if profile not in ("basic", "full"):
 25        raise ValueError("profile must be basic or full")
 26    if type(sample_rows) is not int or not 0 <= sample_rows <= 20:
 27        raise ValueError("sample_rows must be between 0 and 20")
 28    if dataset is not None:
 29        if not isinstance(dataset, str):
 30            raise TypeError("dataset must be str or None")
 31        dataset = dataset.removeprefix("dataset:")
 32        definition = describe(f"dataset:{dataset}")
 33    else:
 34        definition = None
 35    observed = {
 36        "rows": df.height,
 37        "columns": {k: str(v) for k, v in df.schema.items()},
 38        "date_ranges": {},
 39        "dimensions": {},
 40    }
 41    for name, dtype in df.schema.items():
 42        if dtype == pl.Date or isinstance(dtype, pl.Datetime):
 43            observed["date_ranges"][name] = {
 44                "min": df[name].min(),
 45                "max": df[name].max(),
 46            }
 47    for name in ("hw", "maker_name"):
 48        if name in df.columns and (
 49            df.schema[name] in (pl.String, pl.Categorical)
 50            or isinstance(df.schema[name], pl.Enum)
 51        ):
 52            values = df[name].drop_nulls().cast(pl.String).unique().sort()
 53            observed["dimensions"][name] = {
 54                "count": len(values),
 55                "values": values.head(20).to_list(),
 56                "truncated": len(values) > 20,
 57            }
 58    validation = {
 59        "status": "unmatched",
 60        "missing_columns": [],
 61        "extra_columns": [],
 62        "dtype_mismatches": [],
 63        "checks_not_run": [],
 64    }
 65    candidates = []
 66    if definition:
 67        expected = definition["columns"]
 68        validation["missing_columns"] = sorted(set(expected) - set(df.columns))
 69        validation["extra_columns"] = sorted(set(df.columns) - set(expected))
 70        validation["dtype_mismatches"] = [
 71            {
 72                "column": name,
 73                "expected": spec["dtype"],
 74                "observed": str(df.schema[name]),
 75            }
 76            for name, spec in expected.items()
 77            if name in df.columns and str(df.schema[name]) != spec["dtype"]
 78        ]
 79        validation["status"] = (
 80            "schema_mismatch"
 81            if any(
 82                validation[k]
 83                for k in ("missing_columns", "extra_columns", "dtype_mismatches")
 84            )
 85            else "schema_match"
 86        )
 87    else:
 88        for name, spec in DATASETS.items():
 89            overlap = set(df.columns) & set(spec["columns"])
 90            if overlap:
 91                candidates.append(
 92                    {
 93                        "dataset": name,
 94                        "matched_columns": len(overlap),
 95                        "expected_columns": len(spec["columns"]),
 96                        "score": len(overlap)
 97                        / len(set(df.columns) | set(spec["columns"])),
 98                    }
 99                )
100        candidates.sort(key=lambda item: (-item["score"], item["dataset"]))
101    if profile == "full":
102        observed["null_counts"] = dict(zip(df.columns, df.null_count().row(0)))
103        key = definition["key"] if definition else []
104        if key and set(key) <= set(df.columns):
105            keys = df.select(key)
106            validation["key"] = {
107                "columns": key,
108                "duplicate_rows": keys.height - keys.unique().height,
109                "null_rows": keys.filter(pl.any_horizontal(pl.all().is_null())).height,
110            }
111        else:
112            validation["checks_not_run"].append("key: 定義またはキー列がない")
113        if (
114            "hw" in df.columns
115            and df.schema["hw"] == pl.String
116            and "report_date" in df.columns
117            and df.schema["report_date"] == pl.Date
118        ):
119            coverage = (
120                df.group_by("hw")
121                .agg(
122                    pl.len().alias("rows"),
123                    pl.col("report_date").min().alias("begin"),
124                    pl.col("report_date").max().alias("end"),
125                )
126                .sort("hw", nulls_last=True)
127            )
128            observed["coverage_by_hw"] = {
129                "count": coverage.height,
130                "items": coverage.head(20).to_dicts(),
131                "truncated": coverage.height > 20,
132            }
133            if dataset == "hard_sales":
134                dates = (
135                    df.select("hw", "report_date")
136                    .drop_nulls()
137                    .unique()
138                    .sort(["hw", "report_date"])
139                )
140                gaps = dates.with_columns(
141                    pl.col("report_date")
142                    .diff()
143                    .over("hw")
144                    .dt.total_days()
145                    .alias("days_since_previous")
146                ).filter(pl.col("days_since_previous") > 7)
147                observed["weekly_gap_candidates"] = {
148                    "count": gaps.height,
149                    "items": gaps.head(20).to_dicts(),
150                    "truncated": gaps.height > 20,
151                    "meaning": "7日超の間隔。抽出や14日集計でも生じるため欠測とは断定しない",
152                }
153    else:
154        validation["checks_not_run"] = [
155            "null_counts",
156            "key",
157            "coverage_by_hw",
158            "weekly_gap_candidates",
159        ]
160    if sample_rows:
161        observed["sample"] = {
162            "items": df.head(sample_rows).to_dicts(),
163            "truncated": df.height > sample_rows,
164        }
165    return json_value(
166        {
167            "schema_version": SCHEMA_VERSION,
168            "dataset": dataset,
169            "profile": profile,
170            "definition": definition,
171            "observed": observed,
172            "validation": validation,
173            "candidates": candidates[:5],
174            "candidates_truncated": len(candidates) > 5,
175            "unknown": [
176                "データの由来・抽出履歴・DB更新日時はDataFrameから確定できない",
177                "日付の最大値はこのDataFrame内の最新日であり、DBの最新日とは限らない",
178                "列名と型の一致は、列の意味や累計の起点の一致を保証しない",
179            ],
180        }
181    )

def render(result, *, format='markdown'): 
32def render(result, *, format="markdown"):
33    """Return Markdown or strict JSON text; never display or print implicitly."""
34    if format not in ("markdown", "json"):
35        raise ValueError("format must be markdown or json")
36    text = json.dumps(json_value(result), ensure_ascii=False, indent=2, allow_nan=False)
37    if format == "json":
38        return text
39    # Dynamic fences safely display annotation text containing backticks.
40    import re
41
42    longest = max((len(m.group()) for m in re.finditer(r"`+", text)), default=0)
43    fence = "`" * max(3, longest + 1)
44    if not isinstance(result, Mapping):
45        return f"{fence}json\n{text}\n{fence}"
46    lines = []
47    for key, value in json_value(result).items():
48        # The root keys are catalog fields; quote arbitrary keys as inline code.
49        inline_fence = "`" * max(1, longest + 1)
50        key_text = key.replace("\n", " ").replace("\r", " ")
51        lines.append(f"### {inline_fence} {key_text} {inline_fence}")
52        if isinstance(value, str):
53            lines.append(f"{fence}text\n{value}\n{fence}")
54        else:
55            body = json.dumps(value, ensure_ascii=False, indent=2, allow_nan=False)
56            lines.append(f"{fence}json\n{body}\n{fence}")
57    return "\n\n".join(lines)

def annotation_level_guidance( level: int, *, related_hardware: bool = True, in_reporting_window: bool = True, explains_sales_change: bool = False, is_article_topic: bool = False) -> dict:

121    level: int,
122    *,
123    related_hardware: bool = True,
124    in_reporting_window: bool = True,
125    explains_sales_change: bool = False,
126    is_article_topic: bool = False,
127) -> dict:
128    """Return a deterministic editorial decision for one annotation level."""
129    _validate_level(level)
130    for name, value in (
131        ("related_hardware", related_hardware),
132        ("in_reporting_window", in_reporting_window),
133        ("explains_sales_change", explains_sales_change),
134        ("is_article_topic", is_article_topic),
135    ):
136        _validate_flag(name, value)
137
138    band = _band_for(level)
139    relevant = related_hardware and in_reporting_window
140    priority = band["editorial_priority"]
141    if not relevant:
142        should_mention = False
143        reason = "対象ハードまたは記事対象期間に関係しないため"
144    elif priority == "mention_when_relevant":
145        should_mention = True
146        reason = "重要度が高く、対象ハード・期間に関係するため"
147    elif priority == "mention_if_explanatory":
148        should_mention = explains_sales_change or is_article_topic
149        reason = (
150            "販売変化の説明または記事テーマに必要なため"
151            if should_mention
152            else "販売変化の説明にも記事テーマにも必要ではないため"
153        )
154    else:
155        should_mention = is_article_topic
156        reason = (
157            "記事テーマそのものに当たるため"
158            if should_mention
159            else "軽微な出来事であり記事テーマではないため"
160        )
161
162    result = deepcopy(band)
163    result.update(
164        {
165            "level": level,
166            "related_hardware": related_hardware,
167            "in_reporting_window": in_reporting_window,
168            "explains_sales_change": explains_sales_change,
169            "is_article_topic": is_article_topic,
170            "should_consider": relevant and priority != "omit_unless_article_topic",
171            "should_mention": should_mention,
172            "reason": reason,
173        }
174    )
175    return result
